async function sendMessage() { const inputField = document.getElementById('chat-input'); const message = inputField.value.trim(); const modelSelect = document.getElementById('model-select').value; // Se você tiver um checkbox de modo semiológico, pegue o valor, senão use false const medicalMode = document.getElementById('medical-mode') ? document.getElementById('medical-mode').checked : false; if (!message) return; // 1. Mostrar a mensagem do usuário na tela appendMessage('user', message); inputField.value = ''; // 2. Mostrar o balão de "Processando..." const typingIndicator = document.getElementById('typing-indicator'); typingIndicator.style.display = 'block'; try { // 3. Enviar para o Backend (app.py) const response = await fetch('/chat', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ message: message, model: modelSelect, medical_mode: medicalMode }) }); const data = await response.json(); // 4. Esconder o balão de "Processando..." typingIndicator.style.display = 'none'; // 5. Exibir a resposta ou o erro if (response.ok) { appendMessage('athena', data.response); } else { appendMessage('error', '⚠️ Erro: ' + (data.error || 'Falha na comunicação.')); } } catch (error) { typingIndicator.style.display = 'none'; appendMessage('error', '❌ Erro crítico: Não foi possível alcançar o servidor.'); } } function appendMessage(sender, text) { const chatBox = document.getElementById('chat-box'); const msgDiv = document.createElement('div'); msgDiv.style.margin = '10px 0'; msgDiv.style.padding = '10px'; msgDiv.style.borderRadius = '8px'; if (sender === 'user') { msgDiv.style.backgroundColor = '#ff9800'; // Cor do usuário (Laranja) msgDiv.style.color = '#fff'; msgDiv.style.textAlign = 'right'; msgDiv.innerHTML = `Você:
${text}`; } else if (sender === 'error') { msgDiv.style.backgroundColor = '#ff3333'; // Cor de Erro (Vermelho) msgDiv.style.color = '#fff'; msgDiv.innerHTML = `${text}`; } else { msgDiv.style.backgroundColor = '#2d2d2d'; // Cor da IA (Escuro) msgDiv.style.color = '#fff'; // Simples substituição de quebra de linha para ficar bonito const formattedText = text.replace(/\n/g, '
'); msgDiv.innerHTML = `🦉 Athena:
${formattedText}`; } chatBox.appendChild(msgDiv); chatBox.scrollTop = chatBox.scrollHeight; // Desce a rolagem automaticamente }
